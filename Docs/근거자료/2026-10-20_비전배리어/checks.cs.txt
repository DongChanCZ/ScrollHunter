
if(!Application.isPlaying)throw new Exception("Play required");
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
Func<object,string,object> get=(o,n)=>o.GetType().GetField(n,flags).GetValue(o);
Action<object,string,object> set=(o,n,v)=>o.GetType().GetField(n,flags).SetValue(o,v);
Action<object,string,object[]> call=(o,n,a)=>o.GetType().GetMethod(n,flags).Invoke(o,a);
var flow=UnityEngine.Object.FindFirstObjectByType<BattleFlow>();
var deck=UnityEngine.Object.FindFirstObjectByType<DeckSystem>();
var player=UnityEngine.Object.FindFirstObjectByType<Player>();
var cost=UnityEngine.Object.FindFirstObjectByType<CostSystem>();
var info=UnityEngine.Object.FindFirstObjectByType<CombatInfoUI>();
var ui=UnityEngine.Object.FindFirstObjectByType<BattleRewardUI>();
var enemies=UnityEngine.Object.FindFirstObjectByType<EnemyManager>();
var tutorial=UnityEngine.Object.FindFirstObjectByType<TutorialFlow>();
var story=UnityEngine.Object.FindFirstObjectByType<StoryUI>();
var barrier=AssetDatabase.LoadAssetAtPath<SkillData>("Assets/_Project/Data/ScrollHunter/SkillData/Skill_SK15_ArcaneBarrier.asset");
var pool=(BattleRewardOption[])get(flow,"rewardPool");
var option=pool.Single(x=>x.Card==barrier);
var starting=deck.CopyStartingDeck();
var saved=flow.Inheritance; var language=GameText.Current;
bool te=tutorial.enabled,se=story.enabled;
float hold=(float)get(flow,"finishingEffectHold"), countdown=(float)get(flow,"countdownSeconds");
var records=new List<string>(); Action<bool,string> check=(ok,label)=>{if(!ok)throw new Exception(label);records.Add(label);};
string savePath=System.IO.Path.Combine(Application.temporaryCachePath,"ArcaneBarrier-"+Guid.NewGuid().ToString("N")+".json");
Action fresh=()=>{flow.RestartRun();var cards=new List<SkillData>(starting);cards[0]=barrier;deck.SetDeck(cards);typeof(CostSystem).GetProperty("Current").SetValue(cost,5f,null);};
Action<float> step=dt=>call(deck,"AdvanceCast",new object[]{dt});
try {
 tutorial.enabled=false;story.enabled=false;
 set(flow,"inheritance",new InheritanceSave(savePath,pool,starting));
 set(flow,"finishingEffectHold",0f);set(flow,"countdownSeconds",0f);
 check(barrier.Cost==5f&&barrier.CastTime==.2f&&barrier.ShieldAmount==250&&barrier.Category==SkillCategory.Shield&&!barrier.IsChanneling,"data: cost5 cast0.2 shield250");
 check(!starting.Contains(barrier)&&starting.Count==8,"starting deck unchanged");
 check(pool.Count(x=>x.Card!=null)==7&&option.Inheritable&&!string.IsNullOrEmpty(option.InheritanceId),"seventh reward skill with stable inheritance ID");
 fresh();typeof(CostSystem).GetProperty("Current").SetValue(cost,4.999f,null);
 check(!deck.TryUseSlot(0)&&cost.Current==4.999f&&deck.GetHandCard(0)==barrier&&player.Shield==0,"insufficient cost spends nothing");
 fresh();check(deck.TryUseSlot(0),"accepted at exactly 5 cost");
 check(cost.Current==0f&&deck.GetHandCard(0)==starting[4]&&player.Shield==0,"consume once and cycle before effect");
 step(.199f);check(player.Shield==0,"no protection before cast completes");
 step(.0011f);check(player.Shield==250&&deck.IsLocked,"250 shield after 0.2s; normal GCD remains");
 step(.2f);check(!deck.IsLocked,"normal 0.4s action lock");
 var fx=UnityEngine.Object.FindFirstObjectByType<SkillVfx>();
 var binding=((SkillVfx.Entry[])get(fx,"entries")).Single(e=>e.skill==barrier);
 check(binding.selfEffect!=null&&binding.selfEffect.activeSelf,"existing shield animation triggered");
 fresh();player.AddShield(40);check(deck.TryUseSlot(0),"stacking cast accepted");step(.2f);
 check(player.Shield==290,"adds to existing shield");
 float hp=player.CurrentHp;player.TakeDamage(120);
 check(player.Shield==190&&player.CurrentHp==hp,"damage consumes shield before HP");
 player.ApplyStun(1f);check(player.IsStunned&&!player.IsStunImmune,"no added stun immunity");
 fresh();check(deck.TryUseSlot(0),"interruptible defense starts");player.ApplyStun(1f);step(.2f);
 check(player.Shield==0&&cost.Current==0&&!deck.IsCasting,"stun cancels pending shield without refund");
 fresh();deck.TryUseSlot(0);Time.timeScale=0;step(1f);check(player.Shield==0,"pause freezes cast");Time.timeScale=1;step(.2f);check(player.Shield==250,"resume completes cast");
 player.EndBattle();check(player.Shield==0,"battle end clears shield");
 fresh();set(flow,"rewardPool",new[]{option});
 foreach(var enemy in enemies.GetAliveEnemies().ToArray())enemy.TakeDamage(enemy.CurrentHp);
 enemies.NotifyEnemyDied();call(flow,"Update",new object[0]);
 check(flow.State==BattleFlowState.BetweenBattles&&flow.RewardChoiceCount==1&&flow.GetRewardChoice(0)==option,"victory offers barrier");
 check(flow.SelectReward(0)&&flow.SelectReplacementSlot(0)&&flow.ConfirmReplacement(),"reward selection and replacement");
 check(flow.GetDeckCard(0)==barrier&&flow.DeckCount==8&&!option.CanOffer(Enumerable.Range(0,8).Select(flow.GetDeckCard).ToList(),player,cost),"owned skill excluded from rewards");
 flow.NextBattle();check(deck.GetHandCard(0)==barrier&&flow.State==BattleFlowState.Fighting,"acquired skill enters next battle");
 typeof(CostSystem).GetProperty("Current").SetValue(cost,5f,null);check(deck.TryUseSlot(0),"acquired skill usable");step(.2f);check(player.Shield==250,"acquired skill effect correct");
 var inheritance=new InheritanceSave(savePath,pool,starting);
 check(inheritance.TryInherit(option,3),"inherit reward skill");
 var loaded=new InheritanceSave(savePath,pool,starting);
 check(!loaded.LoadFailed&&loaded.BuildDeck()[3]==barrier&&!loaded.CanInherit(option),"inheritance reload and duplicate prevention");
 foreach(GameLanguage lang in Enum.GetValues(typeof(GameLanguage))){
 GameText.SetLanguage(lang);
 string expected=new[]{"비전 배리어","Arcane Barrier","アーケインバリア","奥术屏障","Barrera arcana"}[(int)lang];
 check(barrier.DisplayName==expected,"translated name "+lang);
 var font=Resources.Load<LanguageFontSet>("LanguageFonts").FontFor(lang);
 check(expected.All(c=>char.IsWhiteSpace(c)||font.HasCharacter(c,true,false)),"font coverage "+lang);
 check(info.DescribeCard(barrier).Contains("250")&&info.DescribeCardInfo(barrier).Contains("5"),"description values "+lang);
 }
 check(deck.CopyStartingDeck().SequenceEqual(starting),"serialized starting skills preserved");
} finally {
 flow.ReturnToTitle();set(flow,"rewardPool",pool);set(flow,"inheritance",saved);
 set(flow,"finishingEffectHold",hold);set(flow,"countdownSeconds",countdown);
 tutorial.enabled=te;story.enabled=se;GameText.SetLanguage(language);
}
return Newtonsoft.Json.JsonConvert.SerializeObject(new{passed=records.Count,checks=records,savePath});
